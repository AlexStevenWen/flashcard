function doGet(e) { const p = e.parameter; const action = p.action || 'list'; const folderId = p.folderId; try { const folder = DriveApp.getFolderById(folderId); if (action === 'ping') { return json({ ok:true, version:2, name:'flashcard-backend-sharded' }); } if (action === 'list') { const iter = folder.getFiles(); const files = []; while (iter.hasNext()) { const f = iter.next(); const name = f.getName(); if (name.startsWith('_')) continue; const ext = name.split('.').pop().toLowerCase(); if (!['xlsx','xls','csv'].includes(ext)) continue; files.push({ name, ext, modified: f.getLastUpdated().getTime(), size: f.getSize(), }); } return json({ ok:true, files }); } if (action === 'get') { const name = p.name; const iter = folder.getFilesByName(name); if (!iter.hasNext()) return json({ ok:false, error:'file not found: ' + name }); const f = iter.next(); return json({ ok:true, name, ext: name.split('.').pop().toLowerCase(), modified: f.getLastUpdated().getTime(), size: f.getSize(), base64: Utilities.base64Encode(f.getBlob().getBytes()), }); } if (action === 'getFile') { const name = p.name; const iter = folder.getFilesByName(name); if (!iter.hasNext()) return json({ ok:true, data:null }); return json({ ok:true, data: iter.next().getBlob().getDataAsString() }); } } catch(err) { return json({ ok:false, error: err.message }); } return json({ ok:false, error:'unknown action: ' + action }); } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const folder = DriveApp.getFolderById(body.folderId); if (body.action === 'saveFile') { const iter = folder.getFilesByName(body.name); if (iter.hasNext()) iter.next().setContent(body.data); else folder.createFile(body.name, body.data, 'application/json'); return json({ ok:true }); } } catch(err) { return json({ ok:false, error: err.message }); } return json({ ok:false, error:'unknown action' }); } function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
